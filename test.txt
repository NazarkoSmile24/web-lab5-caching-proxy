ПЕРЕВІРКА ДОВІДКИ
node main.js --help

ЗАПУСК СЕРВЕРА В ОКРЕМОМУ ТЕРМІНАЛІ
npm start -- --host 127.0.0.1 --port 3000 --cache ./cache

ВІДСУТНІСТЬ HOST — очікується код завершення 1
node main.js --port 3000 --cache ./cache
$LASTEXITCODE

ВІДСУТНІСТЬ PORT — очікується код завершення 1
node main.js --host 127.0.0.1 --cache ./cache
$LASTEXITCODE

ВІДСУТНІСТЬ CACHE — очікується код завершення 1
node main.js --host 127.0.0.1 --port 3000
$LASTEXITCODE

НЕПРАВИЛЬНИЙ ПОРТ — очікується код завершення 1
node main.js --host 127.0.0.1 --port abc --cache ./cache
$LASTEXITCODE

НЕДОЗВОЛЕНИЙ МЕТОД — очікується 405 та Allow
curl.exe -i -X POST http://127.0.0.1:3000/200

НЕПРАВИЛЬНИЙ ШЛЯХ — очікується 400
curl.exe -i http://127.0.0.1:3000/abc

СТВОРЕННЯ ПАПКИ ДЛЯ РЕЗУЛЬТАТІВ
New-Item -ItemType Directory -Force results

ПЕРШИЙ GET — очікується 200, image/jpeg, X-Cache: HTTP_CAT
Перед перевіркою файл cache/200.jpg має бути відсутній.
curl.exe -sS -D - http://127.0.0.1:3000/200 -o results/first.jpg

ПОВТОРНИЙ GET — очікується 200, image/jpeg, X-Cache: CACHE
curl.exe -sS -D - http://127.0.0.1:3000/200 -o results/second.jpg

НЕІСНУЮЧА КАРТИНКА — очікується 404
curl.exe -i http://127.0.0.1:3000/999

ВІДСУТНІСТЬ ФАЙЛУ ПІСЛЯ ПОМИЛКИ — очікується False
Test-Path .\cache\999.jpg

PUT ВЛАСНОЇ КАРТИНКИ — очікується 201
Файл own.jpg має бути JPEG-картинкою в корені проєкту.
curl.exe -v --upload-file "./own.jpg" -H "Content-Type: image/jpeg" http://127.0.0.1:3000/418

GET ПІСЛЯ PUT — очікується 200, image/jpeg, X-Cache: CACHE
curl.exe -sS -D - http://127.0.0.1:3000/418 -o results/after-put.jpg

ПЕРЕВІРКА РОЗМІРІВ — усі три значення мають збігатися
Get-Item .\own.jpg, .\cache\418.jpg, .\results\after-put.jpg | Select-Object Name, Length

ПЕРЕВІРКА ЦІЛІСНОСТІ — хеші мають збігатися
Get-FileHash .\own.jpg, .\results\after-put.jpg

ПОПЕРЕДНЯ СПРОБА PUT ЧЕРЕЗ --data-binary
curl.exe -i -X PUT http://127.0.0.1:3000/418 -H "Content-Type: image/jpeg" --data-binary "@own.jpg"
Під час цієї спроби отримано файл розміром 4 байти.
Після перезапуску сервера та повторного PUT через --upload-file
отримано повний файл; розміри та хеші збіглися.
Точну причину попереднього обрізання не встановлено.

ДІАГНОСТИКА ОБРІЗАНОГО ФАЙЛУ
Format-Hex .\results\after-put.jpg

ПІДГОТОВКА КАРТИНКИ ДЛЯ ПЕРЕВІРКИ DELETE — очікується 201
curl.exe -i --upload-file "./own.jpg" -H "Content-Type: image/jpeg" http://127.0.0.1:3000/418

ВИДАЛЕННЯ КАРТИНКИ — очікується 200
curl.exe -i -X DELETE http://127.0.0.1:3000/418

ПЕРЕВІРКА ВІДСУТНОСТІ ФАЙЛУ — очікується False
Test-Path .\cache\418.jpg

ПОВТОРНЕ ВИДАЛЕННЯ — очікується 404
curl.exe -i -X DELETE http://127.0.0.1:3000/418
